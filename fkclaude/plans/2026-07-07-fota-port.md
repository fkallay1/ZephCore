# Port LoRa-FOTA z MeshCore do ZephCore — implementačný plán

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** FOTA delta-patch cez LoRa funkčný na ZephCore repeateri (ProMicro SX1262), so zdieľanými zdrojákmi byte-identickými s MeshCore vetvou.

**Architecture:** Duálne guardy `FOTA_MESHCORE_BUILD`/`FOTA_ZEPHCORE_BUILD` v zdieľaných súboroch; platformové rozdiely koncentrované do shim hlavičiek (`FotaFs.h`, `FotaDebug.h`, `FotaCrypto.h`, `flash_layout.h`). Veľké `.cpp` (Receiver/Patcher) ostávajú platformovo čisté (len premenované typy). Glue vrstva per-projekt (`FotaMyMesh.cpp` ↔ `FotaRepeaterMesh.cpp`). Flasher default beží z RAM (0x20020000), flash cesta ostáva za flagom.

**Tech Stack:** Zephyr 4.4 (west), PlatformIO (MeshCore strana), PSA Crypto + Monocypher (ZephCore), rweather/Crypto (MeshCore), HPatchLite, puff_stream, LittleFS.

**Spec:** `fkclaude/specs/2026-07-07-fota-port-design.md`

```cpp
#pragma once
// =====================================================================
//en: FotaCrypto.h — platform shim for SHA256 + Ed25519 used by FOTA code.
//en: MESHCORE: rweather/Crypto (same dep as mesh::Utils on Arduino).
//en: ZEPHCORE: PSA Crypto (same backend as mesh::Utils) + Monocypher Ed25519.
//sk: FotaCrypto.h — platformový shim pre SHA256 + Ed25519 vo FOTA kóde.
//sk: MESHCORE: rweather/Crypto (rovnaká dep ako mesh::Utils na Arduino).
//sk: ZEPHCORE: PSA Crypto (rovnaký backend ako mesh::Utils) + Monocypher Ed25519.
// =====================================================================
#include <stdint.h>
#include <stddef.h>

#if defined(FOTA_MESHCORE_BUILD)
  #include <SHA256.h>
  #include <Ed25519.h>
  typedef SHA256 FotaSha256;
  static inline bool fota_ed25519_verify(const uint8_t sig[64], const uint8_t pub[32],
                                         const void* msg, size_t len) {
    return Ed25519::verify(sig, pub, msg, len);
  }
#elif defined(FOTA_ZEPHCORE_BUILD)
  #include <psa/crypto.h>
  #include "monocypher-ed25519.h"
  //en: rweather-compatible streaming SHA256 API over PSA
  //sk: streaming SHA256 s rweather-kompatibilným API nad PSA
  class FotaSha256 {
    psa_hash_operation_t _op = PSA_HASH_OPERATION_INIT;
    bool _active = false;
  public:
    FotaSha256() { reset(); }
    ~FotaSha256() { if (_active) psa_hash_abort(&_op); }
    void reset() {
      if (_active) psa_hash_abort(&_op);
      _op = PSA_HASH_OPERATION_INIT;
      _active = (psa_hash_setup(&_op, PSA_ALG_SHA_256) == PSA_SUCCESS);
    }
    void update(const void* data, size_t len) {
      if (_active) psa_hash_update(&_op, (const uint8_t*)data, len);
    }
    void finalize(void* out, size_t len) {
      uint8_t digest[32]; size_t olen = 0;
      if (_active && psa_hash_finish(&_op, digest, sizeof(digest), &olen) == PSA_SUCCESS) {
        for (size_t i = 0; i < len && i < sizeof(digest); i++) ((uint8_t*)out)[i] = digest[i];
      }
      _active = false;
    }
  };
  static inline bool fota_ed25519_verify(const uint8_t sig[64], const uint8_t pub[32],
                                         const void* msg, size_t len) {
    return crypto_ed25519_check(sig, pub, (const uint8_t*)msg, len) == 0;
  }
#else
  #error "FotaCrypto.h: define FOTA_MESHCORE_BUILD or FOTA_ZEPHCORE_BUILD"
#endif
```

```cpp
  #include <string.h>
  //en: /lfs is automounted from DTS fstab; FOTA files live under /lfs/fota/.
  //sk: /lfs automountuje DTS fstab; FOTA súbory žijú pod /lfs/fota/.

  #define FILE_O_WRITE 2

  public:

    void format() {}   //en: shared /lfs — never formatted by FOTA; fota clear removes files

  };
  extern FotaFsClass FotaFS;
  class FotaFile {
    struct fs_file_t _f; bool _open = false;
  public:
    FotaFile(FotaFsClass&) { fs_file_t_init(&_f); }
    ~FotaFile() { close(); }
    bool open(const char* path, int mode) {
      fs_mode_t flags = (mode & FILE_O_WRITE) ? (FS_O_CREATE | FS_O_RDWR) : FS_O_READ;
      _open = (fs_open(&_f, path, flags) == 0);
      if (_open && (mode & FILE_O_WRITE)) fs_seek(&_f, 0, FS_SEEK_END);  //en: append like Adafruit FILE_O_WRITE
      return _open;
    }
    int read(uint8_t* buf, size_t n) { return _open ? (int)fs_read(&_f, buf, n) : -1; }
    int write(const uint8_t* buf, size_t n) { return _open ? (int)fs_write(&_f, buf, n) : -1; }
    bool seek(uint32_t pos) { return _open && fs_seek(&_f, (off_t)pos, FS_SEEK_SET) == 0; }
    uint32_t size() {
      if (!_open) return 0;
      off_t cur = fs_tell(&_f); fs_seek(&_f, 0, FS_SEEK_END);
      off_t end = fs_tell(&_f); fs_seek(&_f, cur, FS_SEEK_SET);
      return (uint32_t)end;
    }
    void close() { if (_open) { fs_close(&_f); _open = false; } }
  };
```

```python
    args = ap.parse_args()
    mc = Path(args.meshcore)
    if not mc.is_dir():
        print(f"[fota_sync] MeshCore nenajdeny: {mc}"); return 2
    diverged, missing = [], []
    for mrel, zrel in SHARED:
        src, dst = mc / mrel, ZEPH_ROOT / zrel
        if not src.is_file(): missing.append(f"MC:{mrel}"); continue
        if not dst.is_file() or not filecmp.cmp(src, dst, shallow=False):
            diverged.append((src, dst))
    for tag in missing: print(f"[fota_sync] CHYBA zdroj: {tag}")
    for src, dst in diverged:
        if args.copy:
            dst.parent.mkdir(parents=True, exist_ok=True)
            shutil.copy2(src, dst); print(f"[fota_sync] copy  {src} -> {dst}")
        else:
            print(f"[fota_sync] DIFF  {dst.relative_to(ZEPH_ROOT)}")
    if not diverged and not missing:
        print(f"[fota_sync] OK — {len(SHARED)} suborov identickych"); return 0
    return 0 if args.copy and not missing else 1

if __name__ == "__main__":
```
